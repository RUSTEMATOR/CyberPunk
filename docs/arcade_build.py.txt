"""Builds webapp/arcade/<game>/ from the cloned open-source games: copy, strip trackers, hook the bridge."""
import re, shutil, pathlib, sys

SRC = pathlib.Path("/tmp/claude-0/oss")
DST = pathlib.Path("/home/claude/nightcity-bot/webapp/arcade/v4")  # versioned path: Cloudflare keeps old files for hours
NC = '<script src="../nc.js" charset="utf-8"></script>'

def fresh(key, src, ignore=()):
    d = DST / key
    if d.exists():
        shutil.rmtree(d)
    shutil.copytree(SRC / src, d, ignore=shutil.ignore_patterns(".git", ".github", "*.md", "node_modules", *ignore))
    for lic in (SRC / src).glob("LICEN*"):
        shutil.copy(lic, d / lic.name)
    return d

def sub(path, old, new, count=1, regex=False):
    s = path.read_text(encoding="utf-8")
    if regex:
        s2, n = re.subn(old, new, s, count=count, flags=re.S)
    else:
        n = s.count(old)
        s2 = s.replace(old, new, count) if n else s
    if not n:
        sys.exit(f"patch miss in {path}: {old[:70]!r}")
    path.write_text(s2, encoding="utf-8")

def head_bridge(index, extra="", bar="bl", viewport=False):
    tag = '<link rel="stylesheet" href="../nc-theme.css">' + NC.replace("<script ", f'<script data-bar="{bar}" ')
    sub(index, "</head>", (VIEWPORT if viewport else "") + tag + extra + "\n</head>")

VIEWPORT = '<meta name="viewport" content="width=device-width,initial-scale=1,maximum-scale=1,user-scalable=no">'
