neon = {"#FF3CAC": (255, 60, 172), "#FFD23F": (255, 210, 63), "#2DE2E6": (45, 226, 230), "#3DFF8F": (61, 255, 143)}
def rgb(t): return f"rgb({t[0]},{t[1]},{t[2]})"
def tint(t): return f"rgb({(t[0] + 255) // 2},{(t[1] + 255) // 2},{(t[2] + 255) // 2})"

a, b = s.index("\twindow.colors = ["), s.index("\twindow.hexagonBackgroundColor")

         "\twindow.hexColorsToTintedColors = {" + ", ".join(f'"{h}": "{tint(t)}"' for h, t in neon.items()) + "};\n"
         "\twindow.rgbToHex = {" + ", ".join(f'"{rgb(t)}": "{h}"' for h, t in neon.items()) + "};\n"
         "\twindow.rgbColorsToTintedColors = {" + ", ".join(f'"{rgb(t)}": "{tint(t)}"' for t in neon.values()) + "};\n\n")
s = s[:a] + block + s[b:]
