import math, random, sys

OUT = sys.argv[1]
SS = 3  # supersampling
BG, PANEL, CELL, LINE = (13, 14, 18), (21, 23, 30), (28, 31, 40), (42, 46, 58)
Y, C, M, V, G, R, T = (255, 210, 63), (45, 226, 230), (255, 60, 172), (138, 92, 255), (61, 255, 143), (255, 77, 94), (232, 232, 238)
FONT_CYR = "/tmp/claude-0/fonts2/tektur-cyrillic-600-normal.woff2"
FONT_LAT = "/tmp/claude-0/fonts2/tektur-latin-600-normal.woff2"

def canvas(w, h, fill=(0, 0, 0, 0)):
    im = Image.new("RGBA", (w * SS, h * SS), fill)
    return im, ImageDraw.Draw(im)

def done(im, w, h, name, glow=0):
    im = im.resize((w, h), Image.LANCZOS)
    if glow:
        g = im.filter(ImageFilter.GaussianBlur(glow))
        im = Image.alpha_composite(g, im)
    im.save(f"{OUT}/{name}")
    return im

def S(*v):
    return [int(round(x * SS)) for x in v]

def a(c, alpha=255):
    return (*c, alpha)

def font(size, cyr=True):
    return ImageFont.truetype(FONT_CYR if cyr else FONT_LAT, size * SS)

def text_center(d, box, s, size, fill, cyr=True, stroke=0, stroke_fill=None):
    f = font(size, cyr)
    x0, y0, x1, y1 = S(*box)
    b = d.textbbox((0, 0), s, font=f, stroke_width=stroke * SS)
    tw, th = b[2] - b[0], b[3] - b[1]
    d.text((x0 + (x1 - x0 - tw) // 2 - b[0], y0 + (y1 - y0 - th) // 2 - b[1]), s, font=f, fill=fill,
           stroke_width=stroke * SS, stroke_fill=stroke_fill)

rnd = random.Random(2077)

# -- the floor module: block, block-perfect, block-rope ------------------------------------------------------------------------
def floor(perfect=False, rope=False):
    w, h0 = 188, 134
    top = 45 if rope else 0
    h = h0 + top
    im, d = canvas(w, h)
    if rope:
        d.line(S(18, top + 2, 94, 4), fill=a(LINE), width=3 * SS)
        d.line(S(170, top + 2, 94, 4), fill=a(LINE), width=3 * SS)
        d.ellipse(S(89, 0, 99, 10), fill=a(Y))
    trim = Y if perfect else M
    d.rectangle(S(4, top + 2, w - 5, top + h0 - 3), fill=a((34, 37, 48)), outline=a(BG), width=2 * SS)
    d.rectangle(S(4, top + 2, w - 5, top + 10), fill=a(trim))
    d.rectangle(S(4, top + h0 - 12, w - 5, top + h0 - 4), fill=a((24, 26, 34)))
    lit = [Y, C, (255, 160, 80), M, (70, 76, 96)]
    for i in range(4):
        x = 16 + i * 42
        col = Y if perfect else rnd.choice(lit)
        d.rectangle(S(x, top + 26, x + 30, top + 82), fill=a(col), outline=a(BG), width=2 * SS)
        d.line(S(x + 15, top + 26, x + 15, top + 82), fill=a(BG), width=2 * SS)
        if col != (70, 76, 96):
            d.rectangle(S(x + 3, top + 29, x + 12, top + 40), fill=a((255, 255, 255), 120))
    d.rectangle(S(20, top + 92, 52, top + 110), fill=a(LINE), outline=a(BG), width=SS)
    d.rectangle(S(136, top + 92, 168, top + 110), fill=a(LINE), outline=a(BG), width=SS)
    d.rectangle(S(64, top + 94, 124, top + 108), fill=a(C if not perfect else Y, 230))
    name = "block-rope.png" if rope else ("block-perfect.png" if perfect else "block.png")
    done(im, w, h, name, glow=2 if perfect else 0)

# -- clouds (smog) and the high-altitude objects ---------------------------------------------------------------------------------------------------
def smog(name, seed):
    r = random.Random(seed)
    im, d = canvas(260, 260)
    for _ in range(7):
        cx, cy, rr = r.randrange(50, 210), r.randrange(110, 160), r.randrange(30, 60)
        d.ellipse(S(cx - rr, cy - rr * 0.7, cx + rr, cy + rr * 0.7), fill=a((70, 50, 110), 150))
    for _ in range(4):
        cx, cy, rr = r.randrange(60, 200), r.randrange(130, 165), r.randrange(25, 45)
        d.ellipse(S(cx - rr, cy - rr * 0.4, cx + rr, cy + rr * 0.4), fill=a(r.choice([M, C]), 70))
    im = im.filter(ImageFilter.GaussianBlur(4 * SS))
    done(im, 260, 260, name)

def satellite(name):
    im, d = canvas(260, 260)
    d.rectangle(S(110, 110, 150, 150), fill=a(LINE), outline=a(C), width=2 * SS)

def debris(name, seed):
    r = random.Random(seed)
    im, d = canvas(260, 260)
    pts = [(130 + math.cos(t) * r.randrange(55, 85), 130 + math.sin(t) * r.randrange(55, 85)) for t in [k * math.pi / 5 for k in range(10)]]
    d.polygon([tuple(S(x, y)) for x, y in pts], fill=a((52, 44, 70)), outline=a(V), width=3 * SS)
    for _ in range(5):
        x, y = r.randrange(95, 165), r.randrange(95, 165)
        d.ellipse(S(x, y, x + 14, y + 10), fill=a((34, 28, 48)))
    d.line(S(90, 150, 170, 110), fill=a(M), width=3 * SS)
    done(im, 260, 260, name, glow=2)

def av(name, body, right=True, lights=None, label=None, cyr=True):
    im, d = canvas(260, 260)
    d.rounded_rectangle(S(40, 110, 220, 160), radius=22 * SS, fill=a(body), outline=a(BG), width=3 * SS)
    nose = 180 if right else 50
    d.polygon(S(nose, 112, nose + (30 if right else -30) + 10, 135, nose, 158) if right else S(nose + 30, 112, 10, 135, nose + 30, 158), fill=a(body))
    d.rounded_rectangle(S(90, 98, 170, 122), radius=10 * SS, fill=a((20, 60, 70)), outline=a(C), width=2 * SS)
    for x in (70, 190) if right else (70, 190):
        d.ellipse(S(x - 18, 156, x + 18, 172), fill=a(C, 200))
    if lights:
        d.rectangle(S(100, 90, 128, 98), fill=a(lights[0]))
        d.rectangle(S(132, 90, 160, 98), fill=a(lights[1]))
    if label:
        text_center(d, (60, 126, 200, 152), label, 16, a(BG) if body == Y else a(T), cyr=cyr)
    done(im, 260, 260, name, glow=3)
